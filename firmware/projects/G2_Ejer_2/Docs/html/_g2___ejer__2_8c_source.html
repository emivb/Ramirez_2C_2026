167static void Mostrar_task(void *pvParameter)
168{
169 while(true)
170 {
171 if (prender)
172 {
173 Leds(distancia);
174 if (!mantener)
175 {
176 LcdItsE0803Write(distancia);
177 }
178 }
179 else
180 {
181 apagartodo();
182 }
183
184 vTaskDelay(mostrar_delay / portTICK_PERIOD_MS);
185 }
186}

188static void Medir_task(void *pvParameter)

193 {
194 distancia = HcSr04ReadDistanceInCentimeters();
