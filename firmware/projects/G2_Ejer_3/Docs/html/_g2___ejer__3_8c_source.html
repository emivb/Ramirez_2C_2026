163void Funcomunicacion(void* param){
164 uint8_t caracter;
165 UartReadByte(UART_PC, &caracter); // acá se busca el dato que avisó la interrupción
166 if (caracter == 'O' || caracter == 'o')
167 {
168 toggle(&prender);
169 UartSendString(UART_PC, "Se apreto la o"); // base 10 -> "1024"
170 UartSendString(UART_PC, "\r\n");
171
172 }
173 if (caracter == 'H' || caracter == 'h')
174 {
175 toggle(&mantener);
176 UartSendString(UART_PC, "Se apreto la H"); // base 10 -> "1024"
177 UartSendString(UART_PC, "\r\n");
178 }
179}

210static void Medir_task(void *pvParameter){
